48bool Board::canPlaceTile(int x, int y, const std::vector<std::vector<int>>& tileShape, char playerSymbol) const {
49 const int tileHeight = static_cast<int>(tileShape.size());
50 const int tileWidth = static_cast<int>(tileShape[0].size());
51
52 if (x < 0 || y < 0 || x + tileHeight > size || y + tileWidth > size) {
53 return false;
54 }
55
56 const bool playerAlreadyOnBoard = playerHasTiles(playerSymbol);
57 bool touchesOwnTerritory = !playerAlreadyOnBoard;
58
59 for (int i = 0; i < tileHeight; ++i) {
60 for (int j = 0; j < tileWidth; ++j) {
61 if (tileShape[i][j] != 1) {
62 continue;
63 }
64
65 const int boardX = x + i;
66 const int boardY = y + j;
67
68 if (grid[boardX][boardY] != '.' || bonuses[boardX][boardY] != LGBonus::None) {
69 return false;
70 }
71
72 if (playerAlreadyOnBoard && !touchesOwnTerritory) {
73 touchesOwnTerritory = isAdjacentToPlayerTerritory(boardX, boardY, playerSymbol);
74 if (!touchesOwnTerritory) {
75 static constexpr int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
76 for (auto dir : dirs) {
77 const int nx = boardX + dir[0];
78 const int ny = boardY + dir[1];
79 if (isInside(nx, ny, size) && grid[nx][ny] == '#' && ownsStoneAt(nx, ny, playerSymbol)) {
80 touchesOwnTerritory = true;
81 break;
82 }
83 }
84 }
85 }
86
87 for (int dx = -1; dx <= 1; ++dx) {
88 for (int dy = -1; dy <= 1; ++dy) {
89 if (dx == 0 && dy == 0) {
90 continue;
91 }
92 const int nx = boardX + dx;
93 const int ny = boardY + dy;
94 if (nx >= 0 && nx < size && ny >= 0 && ny < size) {
95 const bool neighborIsOwnStone = (grid[nx][ny] == '#' && ownsStoneAt(nx, ny, playerSymbol));
96 if (grid[nx][ny] != '.' && grid[nx][ny] != playerSymbol && !neighborIsOwnStone) {
97 return false;
98 }
99 }
100 }
101 }
102 }
103 }
104
105 if (playerAlreadyOnBoard && !touchesOwnTerritory) {
106 return false;
107 }
108
109 return true;
110}

182void Board::placeStartingPoints(const std::vector<char>& playerSymbols) {
183 startCells.clear();
184 auto slots = defaultStartSlots(size);
185 std::size_t slotIndex = 0;
186 std::mt19937 rng(std::random_device{}());
187 std::uniform_int_distribution<int> dist(1, size - 2);
188
189 for (char symbol : playerSymbols) {
190 if (symbol == '\0') {
191 continue;
192 }
193
194 std::pair<int, int> pos{-1, -1};
195 while (slotIndex < slots.size()) {
196 auto candidate = slots[slotIndex++];
197 if (grid[candidate.first][candidate.second] == '.') {
198 pos = candidate;
199 break;
200 }
201 }
202
203 while (pos.first < 0 || grid[pos.first][pos.second] != '.') {
204 const int row = dist(rng);
205 const int col = dist(rng);
206 if (grid[row][col] == '.') {
207 pos = {row, col};
208 }
209 }
210
211 grid[pos.first][pos.second] = symbol;
212 startCells[symbol] = pos;
213 }
214}

216void Board::placeBonusSquares(int numPlayers) {
217 bonuses.assign(size, std::vector<LGBonus>(size, LGBonus::None));
218
219 std::vector<int> available;
220 available.reserve(size * size);
221 for (int r = 0; r < size; ++r) {
222 for (int c = 0; c < size; ++c) {
223 if (grid[r][c] == '.') {
224 available.push_back(r * size + c);
225 }
226 }
227 }
228
229 if (available.empty()) {
230 return;
231 }
232
233 std::mt19937 rng(std::random_device{}());
234 std::shuffle(available.begin(), available.end(), rng);
235
236 const int couponCount = static_cast<int>(std::ceil(numPlayers * 1.5));
237 const int stoneCount = static_cast<int>(std::ceil(numPlayers * 0.5));
238 const int robberyCount = numPlayers;
239
240 size_t cursor = 0;
241 auto placeType = [&](LGBonus bonusType, int count) {
242 for (int i = 0; i < count && cursor < available.size(); ++i) {
243 const int index = available[cursor++];
244 const int row = index / size;
245 const int col = index % size;
246 bonuses[row][col] = bonusType;
247 }
248 };
249
250 placeType(LGBonus::Coupon, couponCount);
251 placeType(LGBonus::Stone, stoneCount);
252 placeType(LGBonus::Robbery, robberyCount);
253}

255std::vector<Board::BonusClaim> Board::claimCompletedBonuses(char playerSymbol) {
256 std::vector<BonusClaim> claims;
257 static constexpr int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
258
259 for (int r = 0; r < size; ++r) {
260 for (int c = 0; c < size; ++c) {
261 LGBonus bonus = bonuses[r][c];
262 if (bonus == LGBonus::None) {
263 continue;
264 }
265
266 bool surrounded = true;
267 for (auto dir : dirs) {
268 const int nr = r + dir[0];
269 const int nc = c + dir[1];
270 if (!isInside(nr, nc, size) || grid[nr][nc] != playerSymbol) {
271 surrounded = false;
272 break;
273 }
274 }
275
276 if (surrounded) {
277 grid[r][c] = playerSymbol;
278 claims.push_back(BonusClaim{bonus, r, c});
279 }
280 }
281 }
282
283 return claims;

340int Board::calculateLargestSquare(char playerSymbol) const {
341 int maxSide = 0;
342 std::vector<std::vector<int>> dp(size, std::vector<int>(size, 0));
343
344 for (int r = 0; r < size; ++r) {
345 for (int c = 0; c < size; ++c) {
346 if (grid[r][c] == playerSymbol) {
347 if (r == 0 || c == 0) {
348 dp[r][c] = 1;
349 } else {
350 dp[r][c] = std::min({dp[r - 1][c], dp[r][c - 1], dp[r - 1][c - 1]}) + 1;
351 }
352 maxSide = std::max(maxSide, dp[r][c]);
353 }
354 }
355 }
356
357 return maxSide * maxSide;
358}
